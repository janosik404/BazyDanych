- imiona i nazwiska nauczycieli ktorzy urodzili sie dokladnie w 1986 roku
	SELECT imie, nazwisko FROM nauczyciel WHERE YEAR(data_urodzenia) = 1986;

- wyswietl imiona i nazwiska oraz daty urodzenia nauczycieli urodzonych po 1990 roku, wyniki posortuj od najmlodszego do najstarszego

	SELECT imie, nazwisko, data_urodzenia FROM nauczyciel 
WHERE YEAR(data_urodzenia) > 1990 
ORDER BY data_urodzenia desc;

- wyswietl nazwy przdmiotow o id ktore n mieszcza sie w przedziale od 5 do 10

- znajdz nauczycieli ktorzy urodzili sie w latach 1970 - 1979(pelne daty)
	SELECT imie, nazwisko FROM nauczyciel 
WHERE data_urodzenia BETWEEN '1970-01-01' AND '1979-12-31' ORDER BY data_urodzenia;

- wyswietl imiona i nazwiska ucznow ktorych imie zaczyna sie na litere a, a nazwisko nie zaczyna sie na litere k
	SELECT imie, nazwisko FROM uczen WHERE imie LIKE 'a%' AND nazwisko NOT LIKE 'k%';
