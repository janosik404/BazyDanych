// wypisz liste ktora bedzie zawierala imie oraz ilosc wystapien tego imiena wsrod uczniow, posortuj alfabetycznie

	SELECT imie, COUNT(*) AS ilosc_wystapien FROM uczen GROUP BY imie ORDER BY ilosc_wystapien DESC;

// lista zawierajaca lata oraz ilosc nauczycieli urodzonych w poszczegolnych latach

	SELECT YEAR(data_urodzenia) AS rok, COUNT(*) FROM nauczyciel GROUP BY rok ORDER BY rok DESC;

// wypisz w poszczegolnych klasach, numer oraz poszczegolne klasy, w raz z iloscia uczniow w tej klasie, posortuj wedlug ilosci uczniow w tej klasie

	SELECT numer, oznaczenie, COUNT(*) AS ilosc_uczniow FROM klasa, uczen WHERE klasa.id = uczen.id GROUP BY id_klasa ORDER BY ilosc_uczniow DESC;
